\section{Introduction}

\subsection{What is \LaTeX{} and why?}

\begin{frame}{Why \LaTeX?}
    \it
    If you merely want to produce a passably good document --- something acceptable and basically readable but not really beautiful --- a simpler system will usually sufﬁce. \\[2\baselineskip]
    With \TeX~the goal is to produce the ﬁnest quality. \\[2\baselineskip]
    ---Donald E. Knuth
\end{frame}

\begin{frame}{Why \LaTeX?}
    \LaTeX{} is a typesetting system (based on \TeX) designed for structured, technical writing.

    \medskip
    For a bachelor's thesis that usually means:
    \begin{itemize}
        \item consistent layout across a long document
        \item strong support for formulas, figures, tables, and references
        \item easy automatic numbering and cross-referencing
        \item source files that stay manageable as the document grows
    \end{itemize}
\end{frame}

\begin{frame}{Why not just use Word?}
    \begin{itemize}
        \item Word is often faster for short, simple documents
        \item \LaTeX{} becomes attractive when you need:
              \begin{itemize}
                  \item many sections, figures, tables, and citations
                  \item stable numbering and references
                  \item precise control over formulas and notation
                  \item collaboration around plain-text source files
              \end{itemize}
        \item It is not WYSIWYG
              \begin{itemize}
                  \item you write source code
                  \item then compile it to PDF
              \end{itemize}
    \end{itemize}
\end{frame}

\begin{frame}{Workshop workflow}
    \begin{itemize}
        \item We will assume you use Overleaf
              \begin{itemize}
                  \item no installation needed
                  \item editor, compiler, and PDF preview in one place
              \end{itemize}
        \item Typical workflow:
              \begin{itemize}
                  \item start from a template or minimal file
                  \item edit the source on the left
                  \item recompile and inspect the PDF on the right
                  \item fix warnings and errors as you go
              \end{itemize}
        \item For this workshop: focus on readable source, not perfect memorisation
    \end{itemize}
\end{frame}

\section{Using \LaTeX{}}

\subsection{The basics}

\begin{frame}[fragile]{A minimal thesis-style document}
    \begin{itemize}
        \item A document has two main parts:
              \begin{itemize}
                  \item the preamble: setup
                  \item the document body: content
              \end{itemize}
        \item Typical starting points:
              \begin{itemize}
                  \item \texttt{\string\documentclass\{article\}}
                  \item \texttt{\string\usepackage\{...\}}
                  \item \texttt{\string\begin\{document\} ... \string\end\{document\}}
              \end{itemize}
        \item For a thesis, the document class is often given by your programme,
              faculty, or publisher
    \end{itemize}
\end{frame}

\subsection{The basics of formatting}

\begin{frame}[fragile]{Commands, environments, and sections}
    \begin{itemize}
        \item Commands start with a backslash:
              \texttt{\string\title\{...\}}, \texttt{\string\textbf\{...\}}
        \item Many commands take arguments in braces
        \item Environments start and end in pairs:
              \texttt{\string\begin\{figure\}} ... \texttt{\string\end\{figure\}}
        \item Structure matters more than appearance:
              \begin{itemize}
                  \item \texttt{\string\section\{Introduction\}}
                  \item \texttt{\string\subsection\{Methods\}}
                  \item \texttt{\string\tableofcontents}
              \end{itemize}
        \item Numbering and contents entries are generated automatically
    \end{itemize}
\end{frame}

\begin{frame}{Text flow and input}
    \begin{itemize}
        \item \LaTeX{} automatically handles:
              \begin{itemize}
                  \item spacing
                  \item paragraph layout
                  \item hyphenation
                  \item numbering of sections, figures, tables, and equations
              \end{itemize}
        \item White space is mostly ignored
        \item An empty line starts a new paragraph
        \item \% starts a comment
    \end{itemize}
\end{frame}

\begin{frame}[fragile]{Special characters and text formatting}
    \begin{itemize}
        \item These characters have a special meaning:
              \texttt{\# \$ \% \^{} \& \_ \{ \} \string~ \string\ }
        \item Use formatting commands instead of manual styling:
              \begin{itemize}
                  \item \texttt{\string\textbf\{...\}} for bold
                  \item \texttt{\string\emph\{...\}} for emphasis
                  \item \texttt{\string\texttt\{...\}} for code-like text
              \end{itemize}
        \item A few common details:
              \begin{itemize}
                  \item quotes are typed as ``like this''
                  \item hyphen, en-dash, em-dash: \texttt{-}, \texttt{--}, \texttt{---}
                  \item \texttt{\string\ldots} gives an ellipsis: \ldots
              \end{itemize}
    \end{itemize}
\end{frame}

\begin{frame}{Formatting commands you will actually use}
    \begin{tabular}{ll}
        \texttt{\string\textbf\{...\}}    & bold text         \\
        \texttt{\string\textit\{...\}}    & italic text       \\
        \texttt{\string\emph\{...\}}      & semantic emphasis \\
        \texttt{\string\underline\{...\}} & underlined text   \\
        \texttt{\string\texttt\{...\}}    & monospaced text   \\
        \texttt{\string\small}            & smaller text size \\
        \texttt{\string\Large}            & larger text size
    \end{tabular}
\end{frame}

\section{Document structure}

\begin{frame}{Document structure for a thesis}
    \begin{itemize}
        \item Front matter often contains:
              \begin{itemize}
                  \item title page
                  \item abstract
                  \item table of contents
              \end{itemize}
        \item Main matter is built from sections and subsections
        \item Back matter may contain appendices and the bibliography
        \item Good structure makes later editing much easier than manual
              formatting does
    \end{itemize}
\end{frame}

\begin{frame}[fragile]{Cross-references}
    \begin{itemize}
        \item Let \LaTeX{} number things for you
        \item Add a label where something is defined:
              \texttt{\string\label\{sec:method\}}
        \item Refer to it later with:
              \begin{itemize}
                  \item \texttt{\string\ref\{sec:method\}}
                  \item \texttt{\string\pageref\{sec:method\}}
              \end{itemize}
        \item This works for sections, equations, figures, and tables
        \item Result: fewer broken references after editing
    \end{itemize}
\end{frame}

\begin{frame}[fragile]{Bibliography}
    \begin{itemize}
        \item Thesis writing almost always means reference management
        \item Keep references in a separate \texttt{.bib} file
        \item Cite in the text with a command such as \texttt{\string\cite\{...\}}
        \item Print the reference list with your chosen bibliography setup
        \item The exact citation style is usually decided by your programme,
              supervisor, or journal
    \end{itemize}
\end{frame}

\section{Lists and tables}

\begin{frame}[fragile]{Lists}
    \begin{itemize}
        \item Three common list environments:
              \begin{itemize}
                  \item \texttt{itemize} for bullet points
                  \item \texttt{enumerate} for numbered steps
                  \item \texttt{description} for term-definition lists
              \end{itemize}
        \item Each new list item starts with \texttt{\string\item}
        \item Lists can be nested
        \item Useful for protocols, assumptions, and summarising results
    \end{itemize}
\end{frame}

\begin{frame}[fragile]{Tables}
    \begin{itemize}
        \item Use \texttt{tabular} to align values in columns
        \item Put important tables in a \texttt{table} environment if they need
              a caption and label
        \item Keep table structure simple and readable
        \item For scientific tables, consistency of units and significant figures
              matters more than decorative lines
    \end{itemize}

    \medskip
    \begin{center}
        \begin{tabular}{l r}
            Sample & Yield (\%) \\
            A      & 81         \\
            B      & 77         \\
            C      & 84
        \end{tabular}
    \end{center}
\end{frame}

\section{Mathematics}

\begin{frame}{Math mode}
    \begin{itemize}
        \item \LaTeX{} has a special mode for mathematical notation
        \item Common forms:
              \begin{itemize}
                  \item inline math with \texttt{\string$...\string$}
                  \item display math with \texttt{\string\[...\string\]}
                  \item numbered equations with \texttt{equation}
              \end{itemize}
        \item Superscripts and subscripts are built in
        \item This matters even in chemistry theses for symbols, variables, and models
    \end{itemize}
\end{frame}

\begin{frame}[fragile]{Examples}
    Inline example:
    $c^2 = a^2 + b^2$

    \medskip
    Displayed and numbered:
    \[
        \lim_{n \to \infty} \sum_{k=1}^n
        \frac{1}{k^2} = \frac{\pi^2}{6}
    \]

    \medskip
    Useful commands include \texttt{\string\frac}, \texttt{\string\sqrt},
    superscripts with \texttt{\string^{}}, and subscripts with
    \texttt{\string_{}}
\end{frame}

\section{Figures and chemistry}

\begin{frame}[fragile]{Figures and floats}
    \begin{itemize}
        \item Use \texttt{\string\usepackage\{graphicx\}} to include images
        \item Put important figures in a \texttt{figure} environment
        \item Add a \texttt{\string\caption} and \texttt{\string\label} so you can
              refer to them later
        \item Let \LaTeX{} help with placement instead of forcing every image by hand
    \end{itemize}

    \medskip
    \begin{figure}
        \centering
        \fbox{\rule{0pt}{2cm}\rule{4cm}{0pt}}
        \caption{Example figure placeholder}
    \end{figure}
\end{frame}

\begin{frame}[fragile]{Chemistry: formulas and reactions}
    \begin{itemize}
        \item A useful package for chemistry is \texttt{mhchem}
        \item It provides \texttt{\string\ce\{...\}} for formulas, charges, and reactions
        \item Examples:
              \begin{itemize}
                  \item \ce{H2O}
                  \item \ce{Fe^{3+}}
                  \item \ce{2 H2 + O2 -> 2 H2O}
                  \item \ce{CH3COOH + H2O <=> CH3COO^- + H3O+}
              \end{itemize}
        \item This is usually clearer and more reliable than hand-formatting chemistry notation
    \end{itemize}
\end{frame}

\begin{frame}[fragile]{Chemistry: units and measurements}
    \begin{itemize}
        \item A useful package for quantities is \texttt{siunitx}
        \item It keeps values and units consistent throughout a thesis
        \item Examples:
              \begin{itemize}
                  \item \texttt{\string\SI\{25\}\{\string\milli\string\liter\}} gives
                        \SI{25}{\milli\liter}
                  \item \texttt{\string\SI\{298.15\}\{\string\kelvin\}} gives
                        \SI{298.15}{\kelvin}
                  \item \texttt{\string\num\{6.022e23\}} gives \num{6.022e23}
              \end{itemize}
        \item This is especially useful in result tables and experimental sections
    \end{itemize}
\end{frame}

              \begin{itemize}

                  \item labels and references instead of manual numbering
