\subsection{Syntax}
\begin{frame}[fragile]{Syntax}
    \begin{itemize}
        \item Designed to focus on writing
        \item Text will be output as text
              \begin{itemize}
                  \item White space is ignored (including newlines)
                  \item Empty line signifies end of paragraph
                  \item \LaTeX{} automagically does hyphenation and outlining
                  \item ``Magical'' characters:

                        \lstinline+# $ % ^ & _ { } ~ \+  %! chktex 14
                  \item Cannot use these in running text
              \end{itemize}
        \item Comments start with a \%
    \end{itemize}
\end{frame}

\begin{frame}[fragile]{Syntax}
    \begin{itemize}
        \item Commands start with a \textbackslash:

              \lstinline+\somecommand[optional]{required}+
        \item Groups start with \{ and end with \}
              \begin{itemize}
                  \item Useful if command influences all following text

                        \lstinline+This {\bf text is bold} but this it not+

                        This {\bf text is bold} but this it not
              \end{itemize}
        \item Environments start with \lstinline+\begin{envname}+ and end with
              \lstinline+\end{envname}+
              \begin{itemize}
                  \item Define new commands that only work in environment
                  \item Change flow of text
                  \item Introduce special elements (i.e. non-text) in
                        document
              \end{itemize}
    \end{itemize}
\end{frame}

\section{Document structure}
\begin{frame}{Document Structure}
    \begin{itemize}
        \item Document consists of two parts:
              \begin{itemize}
                  \item Preamble
                  \item Content
              \end{itemize}
        \item Further forces \LaTeX{}'s focus on content
    \end{itemize}
\end{frame}

\subsection{The preamble}
\begin{frame}{The preamble}
    \begin{itemize}
        \item Set up for the document
              \begin{itemize}
                  \item Documentclass
                  \item Loading packages
                  \item Document wide settings
                  \item New commands/environments
                  \item Meta information (title, author, etc\ldots)
              \end{itemize}
        \item Allows us to change anything about the document
              \begin{itemize}
                  \item Also paper size, margins, etc
                  \item After preamble no longer possible
              \end{itemize}
    \end{itemize}
\end{frame}

\begin{frame}[fragile]{The document class}
    \begin{itemize}
        \item Every document has document class
        \item Specifies general settings like paper size, margins, headers,
              footers, etc\ldots
        \item Is set by \lstinline+\documentclass+ command:

              \lstinline+\documentclass[a4paper, 10pt]{article}+
        \item Different types of document classes:
              \begin{itemize}
                  \item article
                  \item beamer
                  \item report
                  \item book
                  \item letter
                  \item cover-article
                  \item \ldots
              \end{itemize}
    \end{itemize}
\end{frame}

\begin{frame}[fragile]{Importing packages}
    \begin{itemize}
        \item \LaTeX{} files with commands/environments
        \item Can also be used to set setting (i.e. geometry)
        \item Extends \LaTeX{}'s usual capabilities.
        \item Imported with \lstinline+\usepackage+

              \lstinline+\usepackage[hidelinks]{hyperref}+
        \item Contains all kinds of stuff:
              \begin{itemize}
                  \item \lstinline+hyperref+ for making hyperlinks
                  \item \lstinline+graphicx+ for importing images
                  \item \lstinline+amsmath+ for even better math support
                  \item \lstinline+colorx+ for changing colors
                  \item \lstinline+tikz+ for drawing figures in \LaTeX{}
                  \item \lstinline+coffee+ for adding coffee stains
                  \item \lstinline+enigma+ for confusing the allied forces
              \end{itemize}
    \end{itemize}
\end{frame}

\begin{frame}[fragile]{Setting settings}
    \begin{itemize}
        \item Settings depend on packages used and needs of author
        \item \lstinline+\title{}+, \lstinline+\author{}+ and
              \lstinline+\date{}+ are quite common
    \end{itemize}
\end{frame}

\subsection{The content}
\begin{frame}[fragile]{The content}
    \begin{itemize}
        \item Content goes into document environment:
              \begin{lstlisting}
\begin{document}
    ...
\end{document}
            \end{lstlisting}
        \item Normally starts with a title: \lstinline+\maketitle+
        \item For longer documents a table of contents:
              \lstinline+\tableofcontents+
    \end{itemize}
\end{frame}

\begin{frame}[fragile]{Document structure}
    \begin{itemize}
        \item Layout is done by \LaTeX{}, structure is important
        \item Different kinds of headings:
              \begin{itemize}
                  \item \lstinline+\chapter+ only in book and report
                  \item \lstinline+\section+
                  \item \lstinline+\subsection+
                  \item \lstinline+\subsubsection+
                  \item \lstinline+\paragraph+ typeset in running text
                  \item \lstinline+\subparagraph+
                  \item Also a starred version to disable numbering:
                        \lstinline+\section*{}+
              \end{itemize}
        \item \lstinline+\appendix+ changes number to appendix numbering
    \end{itemize}
\end{frame}

\begin{frame}{Text flow}
    \begin{itemize}
        \item \LaTeX{} ignores white space
        \item Works by splitting text up into tokens and paragraphs
        \item Tokens are separated by whitespace or punctuation
        \item Empty line ends paragraph
        \item Sometimes you want to force a space or newline
        \item For space, the non-breaking space: \textasciitilde{}
        \item For newline: \textbackslash\textbackslash{}
              \begin{itemize}
                  \item Warning! This does \emph{not} start a new paragraph
              \end{itemize}
    \end{itemize}
\end{frame}

    \begin{columns}
        \column{0.37\textwidth}
        \begin{itemize}
            \item Changing text size:
                  \begin{itemize}
                      \item \lstinline+\tiny+
                      \item \lstinline+\scriptsize+
                      \item \lstinline+\footnotesize+
                      \item \lstinline+\small+
                      \item \lstinline+\normalsize+
                      \item \lstinline+\large+
                      \item \lstinline+\Large+
                      \item \lstinline+\LARGE+
                      \item \lstinline+\huge+
                      \item \lstinline+\Huge+
                  \end{itemize}
        \end{itemize}
        \column{0.5\textwidth}
        \begin{itemize}
            \item \textbf{Bold text:} \small \lstinline+\textbf{text} {\bf text}+
            \item \textit{Italic text:} \small \lstinline+\textit{text} {\it text}+
            \item \emph{Emphasised text:} \small \lstinline+\emph{text}+
            \item \underline{Underlined text:} \small \lstinline+\underline{text}+
            \item \texttt{Monotype text:} \small \lstinline+\texttt{text} {\tt text}+
        \end{itemize}
    \end{columns}
\end{frame}

\begin{frame}[fragile]{Lists}
    \begin{itemize}
        \item Three kinds of list:
              \begin{enumerate}
                  \item \lstinline+itemize+ bulleted list like the outer list
                  \item \lstinline+enumerate+ numbered list like this one
                  \item \lstinline+description+ list of definitions
              \end{enumerate}
        \item All lists are environments
        \item New items are added with \lstinline+\item+
        \item Lists can be nested up to five levels deep
        \item Numbering happens automatically
    \end{itemize}
\end{frame}

\begin{frame}[fragile]{Referencing}
    \begin{itemize}
        \item \LaTeX{} numbers a lot of stuff
        \item Referring back to parts of document is easy
        \item Name object with \lstinline+\label{name}+
        \item Get number with \lstinline+\ref{name}+
        \item Get the page number of object with \lstinline+\pageref{name}+
        \item (With \texttt{hyperref}) Get name, number, and link
              \lstinline+\autoref{name}+
        \item Can be used to refer to:
              \begin{itemize}
                  \item Sections, subsections, etc\ldots
                  \item Items in an enumeration
                  \item Equations
                  \item Figures
                  \item Tables
              \end{itemize}
    \end{itemize}
\end{frame}

\section{Floats}
\subsection{Fantastical floats and how to use them}
\begin{frame}{What are floats?}
    \begin{itemize}
        \item \LaTeX{} tries to structure text for optimal flow
        \item Does not work if we include figures and tables
        \item Solution: Floats
        \item Float is a container that cannot be broken over pages
        \item \LaTeX{} moves floats through document for optimal placement
        \item Works good for images, tables, pseudocode, and other blocks
        \item Allows us to enrich content
              \begin{itemize}
                  \item Captions
                  \item Subfigures
              \end{itemize}
        \item Can also define our own floats
    \end{itemize}
\end{frame}

\begin{frame}[fragile]{Controlling float placement}
    \begin{itemize}
        \item \LaTeX{} tends to move floats all over the place
        \item Two ways to control floats:
              \begin{itemize}
                  \item Placement specifiers
                        \begin{onlyenv}<2>
                            \begin{itemize}
                                \item Optional argument to the float
                                \item Does not always work ``correctly''
                                \item Pictures still float
                                \item \texttt{h} I want my float here
                                \item \texttt{H} Force float here (requires
                                      \texttt{float} package)
                                \item \texttt{t} I want my float at the top of the page
                                \item \texttt{b} I want my float at the bottom of
                                      the page
                                \item \texttt{!} I really want this
                                \item \texttt{h!} tells \LaTeX{} you really want
                                      the float here
                            \end{itemize}
                        \end{onlyenv}
                  \item Float barriers
                        \begin{onlyenv}<3>
                            \begin{itemize}
                                \item Stops floats from moving past a point
                                \item Useful to keep floats within section
                                \item Require the \texttt{placeins} package
                                \item Created with command \lstinline+\FloatBarrier+
                                \item Automatic float barriers at sections:

                                      \lstinline+\usepackage[section]{placeins}+
                            \end{itemize}
                        \end{onlyenv}
              \end{itemize}
        \item For reports: pictures at end is fine, lots of empty space is
              not
    \end{itemize}
\end{frame}

\begin{frame}[fragile]{Working with images}
    \begin{lstlisting}
\begin{figure}[htb]
    \includegraphics[width=\textwidth]{path/to/file}
    \caption{Something sensible about this image}
    \label{fig:my_image}
\end{figure}
            \end{lstlisting}
    \begin{itemize}
        \item \lstinline+[htb]+ is placement specifier
        \item \lstinline+\includegraphics+ for importing of the image
        \item \lstinline+\caption+ for text below image and numbering
        \item \lstinline+\label+ so we can refer later
    \end{itemize}
\end{frame}

\subsection{Working with floats}

\begin{frame}{Working with floats}
    \begin{itemize}
        \item \LaTeX{} knows the best way to position floats
        \item Bunch of pictures at the end of report not bad
              \begin{itemize}
                  \item Graders know why
                  \item Large sections of white space are more annoying
              \end{itemize}

              \end{itemize}
    \end{itemize}
\end{frame}

\section{Mathematics}

\begin{frame}[fragile]{Mathematics}
    \begin{itemize}
        \item Most of the following commands need the package amsmath
        \item \lstinline+$ ... $+ or \lstinline+\( ... \)+ short hand for inline
              math
        \item \lstinline+$$ ... $$+ or \lstinline+\[ ... \]+ short hand for display math mode
        \item The equation environment displays numbered equations
        \item The align environment is used to arrange equations over multiple
              lines. \lstinline+\\+ denotes a new line and \lstinline+&+ indicates
              a point where the equations should be aligned
        \item Spaces are derived from the expressions, \lstinline+$2-3$+ and
              \lstinline+$2 - 3$+ both print to $2-3$ %! chktex 8
    \end{itemize}
\end{frame}

\begin{frame}[fragile]{Common commands}
    \begin{itemize}
        \item \lstinline|\times| is $\times$, \lstinline|\cdot| is $\cdot$
        \item \lstinline|\sqrt{a}| is $\sqrt{a}$
        \item \lstinline|\frac{a}{b}| is $\frac{a}{b}$
        \item \lstinline+a^{b}+ is $a^b$
        \item \lstinline+a_{b}+ is $a_b$
        \item All greek letters are available. \lstinline+\Delta+ is
              $\Delta$ and \lstinline+\delta+ is $\delta$
        \item Text can be added to the equation with \lstinline+\text{}+.
              \lstinline+2 - 3 \text{ hallo } 3 \times 4+ is $2 - 3 \text{ hallo }
                  3 \times{} 4$
    \end{itemize}
\end{frame}

\begin{frame}[fragile]{Operators}
    \begin{itemize}
        \item \lstinline+\cos (2\pi)+ is $\cos(2\pi)$
        \item \lstinline+\sin^2 (2\theta)+ is $\sin^2(2\theta)$
        \item \lstinline+\tan (2)+ is $\tan(2)$
        \item \lstinline+\lim_{x \to \infty} \exp(-x) = 0+ is
              $\lim_{x \to\infty} \exp(-x) = 0$
        \item \lstinline+x \equiv a \pmod{b}+ is $x \equiv{} a \pmod{b}$
        \item \url{https://en.wikibooks.org/wiki/LaTeX/Mathematics} lists more commands
        \item \url{detexify.kirelabs.org} can identify a latex symbol from a drawing
    \end{itemize}
\end{frame}

\section{Chemistry in TeX}

\begin{frame}[fragile]{Why chemists use TeX}
    \begin{itemize}
        \item Chemistry writing needs consistent notation for formulas, reactions, units, and references
        \item \LaTeX{} is strong at keeping that notation uniform across long reports and theses
        \item Two packages show up all the time in chemistry documents:
              \begin{itemize}
                  \item \lstinline+mhchem+ for chemical formulas and reactions
                  \item \lstinline+siunitx+ for numbers, units, and uncertainties
              \end{itemize}
        \item The goal is readable source and publication-quality output
    \end{itemize}
\end{frame}

\begin{frame}[fragile]{Chemical formulas and reactions}
    \begin{itemize}
        \item \lstinline+\ce+ formats formulas, ions, and reactions
        \item Examples:
              \begin{itemize}
                  \item Water: \lstinline|H2O|: \ce{H2O}
                  \item Oxidation state: \lstinline|Fe\^{}\{2+\}|: \ce{Fe^{2+}}
                  \item Reaction: \lstinline|2 H2 + O2 -> 2 H2O|: \ce{2 H2 + O2 -> 2 H2O}
                  \item Equilibrium: \lstinline|CH3COOH + H2O <=> CH3COO\^{}- + H3O+|: \\
                        \ce{CH3COOH + H2O <=> CH3COO^- + H3O+}
              \end{itemize}
        \item This keeps subscripts, charges, arrows, and stoichiometry consistent
    \end{itemize}
\end{frame}

\begin{frame}[fragile]{Units, quantities, and data}
    \begin{itemize}
        \item \lstinline+siunitx+ keeps units separated from values
        \item Examples:
              \begin{itemize}
                  \item Volume: \lstinline|\SI{25}{\milli\liter}|: \SI{25}{\milli\liter}
                  \item Temperature: \lstinline|\SI{298.15}{\kelvin}|: \SI{298.15}{\kelvin}
                  \item Concentration: \lstinline|\SI{1.20e-3}{\mole\per\liter}|: \SI{1.20e-3}{\mole\per\liter}
                  \item Count with formatting: \lstinline|\num{6.022e23}|: \num{6.022e23}
              \end{itemize}
        \item Useful for lab reports, tables, and plots where significant figures matter
    \end{itemize}
\end{frame}

\begin{frame}[fragile]{Common chemistry document tips}
    \begin{itemize}
        \item Put reaction schemes, spectra, and tables in floats so they can be captioned and referenced
        \item Use \lstinline+\label+ and \lstinline+\ref+ for compounds, schemes, and figures
        \item Keep citations in a \lstinline+.bib+ file so experimental sources and papers stay manageable
        \item If your report has a lot of notation, define a few small commands in the preamble to reduce repetition
    \end{itemize}
\end{frame}

\section{Bibtex}
\begin{frame}[fragile]{Bibtex}
    Useful package for referencing sources.
    \begin{itemize}
        \item Stores references in seperate ".bib" file.
              \begin{itemize}
                  \item Set style with \lstinline+\bibliographystyle{plain}+
                  \item Include bib file with \lstinline+\bibliography{filename}+
                  \item These should be placed in this order at the location
                        you want the reference list to be printed
              \end{itemize}
