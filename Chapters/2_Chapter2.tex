\subsection{Referring to the figure}
Here we reference figure \ref{fig:2a} and \ref{fig:2b}, but we can also reference like this: \autoref{fig:2}. 

%Figures
\begin{figure}[h]
  \centering
  \begin{subfigure}[b]{0.49\textwidth}
    \centering
    \includegraphics[width=\textwidth]{Images/RUGLogo.png}
    \caption{Left figure}
    \label{fig:2a}
  \end{subfigure}
  \begin{subfigure}[b]{0.49\textwidth}
    \centering
    \includegraphics[width=\textwidth]{Images/RUGLogo.png}

    \label{fig:2b}

  \caption{Two figures next to each other.}
  \label{fig:2}
